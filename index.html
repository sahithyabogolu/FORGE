// 1. DYNAMIC SCALE DETECTOR ("in thousands", "in millions") function extractScaleMultiplier(text) { const t = text.toLowerCase().slice(0, 3000); // Check filing header block if (t.includes("in millions") || t.includes("millions of dollars") || t.includes("($ in millions)")) { return 1000000; } if (t.includes("in thousands") || t.includes("thousands of dollars") || t.includes("($ in thousands)")) { return 1000; } return 1; } // 2. PATCHED METRIC EXTRACTOR function extractMetric(text, metricType) { const lines = text.split(/\r?\n/); const scaleMultiplier = extractScaleMultiplier(text); // Exclusions and keyword patterns const config = { revenue: { include: [/\btotal revenue\b/i, /\brevenues\b/i, /\btotal net sales\b/i, /\brevenue\b/i], exclude: [/cost of/i, /deferred/i, /unearned/i] }, expenses: { include: [/\btotal operating expenses\b/i, /\btotal expenses\b/i, /\btotal cost of sales\b/i], exclude: [/prepaid/i, /accrued/i, /tax expense/i] }, cash: { include: [/\bcash and cash equivalents\b/i, /\btotal cash\b/i, /\bcash and short-term investments\b/i], exclude: [/provided by/i, /used in/i, /flow/i] }, debt: { include: [/\btotal debt\b/i, /\blong-term debt\b/i, /\bnotes payable\b/i, /\btotal liabilities\b/i], exclude: [/interest expense/i, /debt issuance/i] } }; const target = config[metricType]; if (!target) return 0; for (const line of lines) { const trimmed = line.trim(); if (!trimmed) continue; // Check exclusions if (target.exclude.some(rx => rx.test(trimmed))) continue; // Check inclusions if (target.include.some(rx => rx.test(trimmed))) { // Regex for numbers with optional negative parens e.g., (1,234.56) or 1,234.56 const numberRegex = /(\(?\$\s*[\d,]+(?:\.\d+)?\)?)/g; const matches = trimmed.match(numberRegex); if (matches && matches.length > 0) { // Grab FIRST column match (Current Period) instead of right-most (Prior Period) let rawStr = matches[0].trim().replace(/\$/g, ""); let isNegative = false; if (rawStr.startsWith("(") && rawStr.endsWith(")")) { isNegative = true; rawStr = rawStr.slice(1, -1); } const numericValue = parseFloat(rawStr.replace(/,/g, "")); if (!isNaN(numericValue)) { const finalVal = (isNegative ? -numericValue : numericValue) * scaleMultiplier; return finalVal; } } } } return 0; } // 3. REVISED INVESTOR RISK AUDIT WITH CONTINUOUS SCORING & DEBT METRICS async function analyzeInvestor() { const f = $("tenkFile").files[0]; if (!f) return alert("Select a filing document first."); const text = await readFileText(f); // Populate state with fixed extractors state.rev = extractMetric(text, "revenue"); state.burn = extractMetric(text, "expenses"); state.cash = extractMetric(text, "cash"); state.debt = extractMetric(text, "debt"); const netIncome = state.rev - state.burn; const opMargin = state.rev > 0 ? (netIncome / state.rev) * 100 : -100; // Runway Math: Monthly Net Burn = (Operating Expenses - Revenue) / 12 const monthlyNetBurn = (state.burn - state.rev) / 12; let runwayMonths = 999; if (monthlyNetBurn > 0) { runwayMonths = state.cash / monthlyNetBurn; } // Debt-to-Cash Ratio const debtToCash = state.cash > 0 ? state.debt / state.cash : (state.debt > 0 ? 99 : 0); // FLAG DETECTION const flags = []; if (opMargin < 0) { flags.push(`Negative Margin (${opMargin.toFixed(1)}%): Operating expenses exceed total revenue.`); } if (runwayMonths < 12 && monthlyNetBurn > 0) { flags.push(`Critical Runway (${runwayMonths.toFixed(1)} Mos): Liquidity exhausts quickly at current net burn rate.`); } if (debtToCash > 2.0) { flags.push(`High Leverage (Debt/Cash ${debtToCash.toFixed(1)}x): Total debt significantly exceeds liquid cash reserves.`); } if (state.rev === 0) { flags.push("Zero Revenue Reported: Filing indicates no active commercial top-line generation."); } // CONTINUOUS WEIGHTED SCORING ENGINE (0 to 100) // Base Score: 100 points let score = 100; // Margin Penalty: Deduct linearly for negative operating margins up to -40 pts if (opMargin < 0) { const marginPenalty = Math.min(40, Math.abs(opMargin) * 0.8); score -= marginPenalty; } // Runway Penalty: Deduct up to -40 pts if runway is under 18 months if (runwayMonths < 18) { const runwayPenalty = Math.min(40, (18 - runwayMonths) * 2.2); score -= runwayPenalty; } // Debt Penalty: Deduct up to -20 pts based on debt ratio if (debtToCash > 1.0) { const debtPenalty = Math.min(20, (debtToCash - 1.0) * 10); score -= debtPenalty; } score = Math.max(0, Math.min(100, Math.round(score))); // VERDICT DETERMINATION BASED ON SCORING let verdict = "INVESTMENT SUITABLE"; if (score < 40) { verdict = "DO NOT INVEST"; } else if (score < 70) { verdict = "HIGH CAUTION"; } // DOM UPDATES $("vVerdict").textContent = verdict; $("vScore").textContent = `${score}/100`; $("vMargin").textContent = `${opMargin.toFixed(1)}%`; $("vRunway").textContent = monthlyNetBurn > 0 ? `${runwayMonths.toFixed(1)} Mos` : "Sustainable"; const rList = $("redFlagsList"); rList.innerHTML = ""; if (flags.length > 0) { flags.forEach(flg => { const li = document.createElement("li"); li.textContent = flg; rList.appendChild(li); }); $("redFlagsBox").style.display = "block"; } else { $("redFlagsBox").style.display = "none"; } $("screenerOutput").style.display = "block"; }
